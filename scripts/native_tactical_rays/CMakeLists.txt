cmake_minimum_required(VERSION 3.14)
project(tactical_rays LANGUAGES CXX)
add_executable(tactical_rays main.cpp)
add_executable(tactical_field_rays field.cpp)
add_library(tactical_reference_cast SHARED reference_cast.cpp)
add_library(tactical_floor_atlas SHARED floor_atlas.cpp floor_cone.cpp reference_cast.cpp)
foreach(target tactical_rays tactical_field_rays tactical_reference_cast tactical_floor_atlas)
  target_compile_features(${target} PRIVATE cxx_std_17)
  target_include_directories(${target} PRIVATE ../../native/height)
  if(MSVC)
    target_compile_options(${target} PRIVATE /O2 /fp:strict /EHsc)
  else()
    target_compile_options(${target} PRIVATE -O2 -fno-fast-math -ffp-contract=off)
  endif()
endforeach()
